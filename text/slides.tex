                5: \norm{\pmb{a}_t} (\mu f_n - \norm{\pmb{f}_t}) = 0 \\
                6: \norm{\pmb{a}_t} \norm{\pmb{f}_t} = - \langle \pmb{a}_t, \pmb{f}_t \rangle
            \end{cases}
            \]
        \end{column}

        \begin{column}{0.05\textwidth}
        \end{column}

        \begin{column}{0.7\textwidth}
            $f_n$ -- величина силы реакции опоры (скаляр)
            
            \medskip
            $a_n$ -- относительное ускорение в направлении нормали (скаляр)
            
            \medskip
            $\pmb{f}_t$ -- сила трения (2D вектор в плоскости контакта)
            
            \medskip
            $\pmb{a}_t$ -- относительное ускорение в плоскости контакта (тоже 2D вектор)
        \end{column}
    \end{columns}

    \begin{enumerate}
        \item Сила реакции опоры применяется в направлении отталкивания
        \item Нет проникновения одного тела в другое
        \item Если тела отдаляются, сила в точке контакта отсутствует
        \item Сила трения ограничена силой реакции опоры
        \item Если есть проскальзывание, то сила трения максимальна по величине
        \item Сила трения направлена против ускорения (если оно есть)
    \end{enumerate}
\end{frame}

\begin{frame}
\small
\frametitle{Формализация. Как силы влияют на ускорение?}
    
    Хотим узнать, как влияет какая-то сила на ускорение какой-то точки тела.

    \begin{block}{Что известно}
        $
        \begin{aligned}
            \pmb{c} &- \text{центр масс тела } (\pmb{c} \in \mathbb{R}^3) \\
            \pmb{p}_1 &- \text{точка, в которой применяется сила } (\pmb{p}_1 \in \mathbb{R}^3) \\
            \pmb{p}_2 &- \text{интересующая нас точка } (\pmb{p}_2 \in \mathbb{R}^3) \\
            \pmb{f} &- \text{применяемая сила } (\pmb{f} \in \mathbb{R}^3) \\
            m, I &- \text{масса тела и его тензор инерции } (m \in \mathbb{R}, I \in \mathbb{M}_{3 \times 3})
        \end{aligned}
        $
    \end{block}

    \begin{block}{Вычисляем}
        $\pmb{a}_{lin} = \pmb{f}/m$ -- линейный вклад силы в ускорение
        
        $\pmb{r}_1 = \pmb{p}_1 - \pmb{c}$
        
        $\pmb{r}_2 = \pmb{p}_2 - \pmb{c}$

        $\pmb{\tau} = \pmb{r}_1 \times \pmb{f}$ -- крутящий момент

        $\pmb{a}_{ang} = I^{-1} \pmb{\tau} \times \pmb{r}_2$ -- угловой вклад силы в ускорение.

        $\pmb{a} = \pmb{a}_{lin} + \pmb{a}_{ang}$
    \end{block}
\end{frame}

\newcommand*{\defeq}{\stackrel{\scalebox{0.5}{\text{def}}}{:=}}

\begin{frame}
\frametitle{Формализация. Проецируем все силы}
        \begin{columns}
        \begin{column}[T]{0.5\textwidth}
        \[    
        \pmb{f} \defeq \norm{\pmb{f}} \cdot \pmb{\hat{f}} \defeq f \cdot \pmb{\hat{f}}
        \]
        
        \[
        \pmb{a}_{lin} = (\pmb{\hat{f}}/m) \cdot f        
        \]
        
        \[
        \pmb{\tau} = \pmb{r}_1 \times (f \cdot \pmb{\hat{f}}) = (\pmb{r}_1 \times \pmb{\hat{f}}) \cdot f
        \]
        \begin{align*} 
            \pmb{a}_{ang} &= I^{-1} \pmb{\tau} \times \pmb{r}_2 \\
            &= I^{-1} ((\pmb{r}_1 \times \pmb{\hat{f}}) \cdot f) \times \pmb{r}_2 \\
            &= (I^{-1} (\pmb{r}_1 \times \pmb{\hat{f}}) \times \pmb{r}_2) \cdot f
        \end{align*}
        \end{column}

        \begin{column}[T]{0.5\textwidth}
        Отображение линейное!

        \[A
        \begin{pmatrix} f_1 \\ f_2 \\ \vdots \\ f_{N} \end{pmatrix}
        =
        \begin{pmatrix} a_1 \\ a_2 \\ \vdots \\ a_N \end{pmatrix}
        \]
        \end{column}
        \end{columns}
\end{frame}

\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small
    $
    \begin{cases}
        f_n \ge 0 \\
        a_n \ge 0 \\

        \norm{\pmb{a}_t} \norm{\pmb{f}_t} = - \langle \pmb{a}_t, \pmb{f}_t \rangle
    \end{cases}
    $

    \bigskip
    Условие дополняющей нежесткости:
    $0 \le a \perp b \ge 0  \iff  \{ a \ge 0, b \ge 0, ab = 0 \}$

\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n \perp a_n \ge 0 \\
        \mu f_n \ge \norm{\pmb{f}_t} \\
        \norm{\pmb{a}_t} (\mu f_n - \norm{\pmb{f}_t}) = 0 \\
        \norm{\pmb{a}_t} \norm{\pmb{f}_t} = -\langle \pmb{a}_t, \pmb{f}_t \rangle
    \end{cases}
    $

    \bigskip
    Зафиксируем направление силы трения.
    
    \bigskip
    $
    \begin{aligned}
    \pmb{f}_t &= \norm{\pmb{f}_t} \cdot \pmb{\hat{f}}_t = f_t \cdot \pmb{\hat{f}}_t \\
    \pmb{\hat{f}}_t &= const
    \end{aligned}
    $
\end{frame}


\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n \perp a_n \ge 0 \\
        \mu f_n \ge |f_t| \\
        \norm{\pmb{a}_t} (\mu f_n - |f_t|) = 0 \\
        \norm{\pmb{a}_t} |f_t| = -\langle \pmb{a}_t, \pmb{\hat{f}}_t \rangle \cdot f_t
    \end{cases}
    $

    \bigskip
    В решении этой системы сила трения будет направлена против относительного ускорения в плоскости контакта.
    Допустим, что это условие выполнено:
    
    \medskip
    $
    \begin{aligned}
    \pmb{a}_t &= \norm{\pmb{a}_t} \cdot \pmb{\hat{a}}_t = a_t \cdot \pmb{\hat{a}}_t \\
    \pmb{\hat{a}}_t &= \pmb{\hat{f}}_t = const
    \end{aligned}
    $

    \medskip
    Тогда нет смысла рассматривать относительное ускорение в плоскости, будем рассматривать только его проекцию на направление силы трения.
    Из-за сделанного допущения проекция на ортоганальное направление будет нулевой.

\end{frame}



\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n \perp a_n \ge 0 \\
        \mu f_n \ge |f_t| \\
        |a_t| (\mu f_n - |f_t|) = 0 \\
        |a_t| |f_t| = -a_t \cdot f_t
    \end{cases}
    $

    \bigskip
    Вместо силы $f_t$ будем прикладывать \textbf{либо} $f_{t+}$, \textbf{либо} $f_{t-}$, направленные в противоположных направлениях.
    
    \bigskip
    Поэтому потребуем $0 \le f_{t+} \perp f_{t-} \ge 0$
    
    \bigskip
    $\begin{aligned}
    \text{Тогда } f_t& \text{ можно записать как: }& f_t &= f_{t+} - f_{t-} \\
    \text{А } |f_t|& \text{ можно записать как: }& |f_t| &= f_{t+} + f_{t-}
    \end{aligned}$

    \bigskip
    На множество значений, которое может принимать $f_t$ это не повлияет.
\end{frame}



\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n \perp a_n \ge 0 \\
        \mu f_n \ge f_{t+} + f_{t-} \\
        |a_t| (\mu f_n - f_{t+} - f_{t-}) = 0 \\
        |a_t| (f_{t+} + f_{t-}) = -a_t \cdot (f_{t+} - f_{t-}) \\
        0 \le f_{t+} \perp f_{t-} \ge 0
    \end{cases}
    $

    \bigskip
    Введем вспомогательную переменную $\beta$ -- индикатор наличия движения.

    \medskip
    $
    \begin{cases}
        0 \le a_t + \beta \perp f_{t+} \ge 0 \\
        0 \le -a_t + \beta \perp f_{t-} \ge 0
    \end{cases}
    $

\end{frame}


\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n \perp a_n \ge 0 \\
        \mu f_n \ge f_{t+} + f_{t-} \\
        |a_t| (\mu f_n - f_{t+} - f_{t-}) = 0 \\
        |a_t| (f_{t+} + f_{t-}) = -a_t \cdot (f_{t+} - f_{t-}) \\
        0 \le f_{t+} \perp f_{t-} \ge 0 \\
        \\
        0 \le a_t + \beta \perp f_{t+} \ge 0 \\
        0 \le -a_t + \beta \perp f_{t-} \ge 0 \\
        \beta \ge 0
    \end{cases}
    $

    \bigskip
    Условия с индикатором из двух сил разрешают только одну силу: действующую против относительного ускорения. Таким образом, условия 4 и 5 становятся лишними.

\end{frame}


\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n \perp a_n \ge 0 \\
        \mu f_n \ge f_{t+} + f_{t-} \\
        |a_t| (\mu f_n - f_{t+} - f_{t-}) = 0 \\
        \\ \\ \\
        0 \le a_t + \beta \perp f_{t+} \ge 0 \\
        0 \le -a_t + \beta \perp f_{t-} \ge 0 \\
        \beta \ge 0
    \end{cases}
    $ 

    \bigskip
    $
    \begin{aligned}
    (\mu f_n \ge f_{t+} + f_{t-}) &\iff (\mu f_n - f_{t+} - f_{t-} \ge 0) \\
    (|a_t| \ne 0) &\iff (\beta \ne 0)
    \end{aligned}
    $

\end{frame}


\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le f_n &\perp a_n \ge 0 \\
        0 \le a_t + \beta &\perp f_{t+} \ge 0 \\
        0 \le -a_t + \beta &\perp f_{t-} \ge 0 \\
        \\
        \mu f_n - f_{t+} - f_{t-} &\ge 0  \\
        \beta &\ge 0 \\
        \beta (\mu f_n - f_{t+} - f_{t-}) &= 0
    \end{cases}
    $ 
\end{frame}



\begin{frame}[t]
\frametitle{Формализация. Как решать?}
    \small \bigskip \bigskip \medskip
    $
    \begin{cases}
        0 \le a_n                       &\perp f_n    \ge 0 \\
        0 \le a_t + \beta               &\perp f_{t+} \ge 0 \\
        0 \le -a_t + \beta              &\perp f_{t-} \ge 0 \\
        0 \le \mu f_n - f_{t+} - f_{t-} &\perp \beta  \ge 0
    \end{cases}
    $

    \bigskip
    Готово. Свели задачу к LCP (Linear Complementarity Problem).
    Решаем её алгоритмом Лемке.
    
    \begin{block}{Формулировка LCP}
    Найти такие $x$, что:
    $
    \begin{cases}
        x \ge 0 \\
        Ax + b \ge 0 \\
        x^T(Ax+b) = 0
    \end{cases}
    $
    \end{block}
\end{frame}


\begin{frame}[t]
\frametitle{LCP}
    $
    \begin{matrix}
        A & x & & b \\
        \begin{pmatrix}
            A_{nn} & A_{nt} & -A_{nt} & 0 \\
            A_{tn} & A_{tt} & -A_{tt} & E \\
            -A_{tn} & -A_{tt} & A_{tt} & E \\
            M & -E & -E & 0
        \end{pmatrix} &
        \begin{pmatrix}
        F_{n} \\
        F_{t+} \\
        F_{t-}  \\
        B
        \end{pmatrix}
        & + &
        \begin{pmatrix}
        P_{n} \\
        P_{t} \\
        -P_{t}  \\
        0
        \end{pmatrix} & \ge 0
    \end{matrix} 
    $

    \medskip
    $
    \begin{matrix}
        M & F_n & F_{t+} & F_{t-} & B \\

        \begin{pmatrix} \mu_1 \\ & \ddots \\ & & \mu_N \end{pmatrix} &
        \begin{pmatrix} f_{n,1} \\ \vdots \\ f_{n,N} \end{pmatrix} &
        \begin{pmatrix} f_{t+,1} \\ \vdots \\ f_{t+,N} \end{pmatrix} &
        \begin{pmatrix} f_{t-,1} \\ \vdots \\ f_{t-,N} \end{pmatrix} &
        \begin{pmatrix} \beta_{1} \\ \vdots \\ \beta_N \end{pmatrix}
    \end{matrix}
    $

    \medskip
    $A_{nn}$ -- проекция сил реакции опоры на направления нормалей плоскостей касаний.

    $A_{nt}$, $A_{tn}$, $A_{tt}$ -- аналогично, проекции сил на направления.

    $P_{n}$, $P_{t}$ -- проекции внешних сил на направления.
\end{frame}


\begin{frame}
    \small
    \frametitle{Выбор лучшего решения}
    
    Перебираем различные направления сил трения.

    Для сведения системы к решаемой мы сделали одно допущение:
    \begin{quote}
    В решении этой системы сила трения будет направлена против относительного ускорения в плоскости контакта.
    Допустим, что это условие выполнено.
    \end{quote}

    Потенциально нарушаемые условия:

    \[
    \begin{cases}
        \norm{\pmb{a}_t} (\mu f_n - \norm{\pmb{f}_t}) = 0 & \text{Штраф: } |\norm{a_t} (\mu f_n - \norm{f_t})| \\
        \norm{\pmb{a}_t} \norm{\pmb{f}_t} = - \langle \pmb{a}_t, \pmb{f}_t \rangle & \text{Штраф: } |\norm{a_t} \norm{f_t} + (a_t \cdot f_t)|
    \end{cases}
    \]

    \bigskip
    Выбираем решение с наименьшим штрафом.

\end{frame}
